outlier_bounds = {}

for column in data_selected.columns:
    # Calculate Q1 (25th percentile)
    Q1 = data_selected[column].quantile(0.25)
    # Calculate Q3 (75th percentile)
    Q3 = data_selected[column].quantile(0.75)
    # Calculate IQR
    IQR = Q3 - Q1
    # Calculate Lower Bound
    lower_bound = Q1 - 1.5 * IQR
    # Calculate Upper Bound
    upper_bound = Q3 + 1.5 * IQR
    
    outlier_bounds[column] = {
        'Q1': Q1,
        'Q3': Q3,
        'IQR': IQR,
        'lower_bound': lower_bound,
        'upper_bound': upper_bound
    }

print("Calculated outlier bounds for each column:")
for column, bounds in outlier_bounds.items():
    print(f"Column: {column}")
    for key, value in bounds.items():
        print(f"  {key}: {value:.2f}")
